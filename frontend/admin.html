Server-side Implementation

Node.js · Express — auto-included in our CMS middleware

// packages/api/services/capi.ts
import crypto from 'crypto';
import axios from 'axios';

const PIXEL_ID = process.env.FB_PIXEL_ID;
const ACCESS_TOKEN = process.env.FB_CAPI_TOKEN;

const sha256 = (v) => crypto
  .createHash('sha256')
  .update(String(v).toLowerCase().trim())
  .digest('hex');

export async function sendEvent(event, req) {
  const payload = {
    data: [{
      event_name: event.name,              // 'Purchase'
      event_time: Math.floor(Date.now()/1000),
      event_id:   event.id,                // dedupe w/ Pixel
      action_source: 'website',
      event_source_url: event.url,
      user_data: {
        em:         event.email ? [sha256(event.email)] : undefined,
        ph:         event.phone ? [sha256(event.phone)] : undefined,
        external_id: event.userId ? [sha256(event.userId)] : undefined,
        client_ip_address: req.ip,
        client_user_agent: req.get('user-agent'),
        fbp: req.cookies._fbp,
        fbc: req.cookies._fbc
      },
      custom_data: {
        currency: 'BDT',
        value:    event.value,
        content_ids: event.contentIds,
        content_type: 'product'
      }
    }],
    test_event_code: process.env.FB_TEST_CODE
  };

  return axios.post(
    `https://graph.facebook.com/v19.0/${PIXEL_ID}/events`,
    payload,
    { params: { access_token: ACCESS_TOKEN } }
  );
}
